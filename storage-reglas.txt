rules_version = '2';

// Reglas de Firebase Storage (consola → Storage → Rules → pegar esto → Publicar).
// Videos de los informes de turno: los sube quien carga el informe (sin clave, igual que el
// informe), no se pueden pisar ni borrar, y solo los ve la cuenta de Gerencia.
service firebase.storage {
  match /b/{bucket}/o {
    match /informes/{turno}/{ts}/{archivo} {
      allow read: if request.auth != null && request.auth.token.email == 'gerencia@candela-app.com';
      allow create: if resource == null
                    && request.resource.size < 160 * 1024 * 1024
                    && request.resource.contentType.matches('video/.*');
    }
    // Todo lo demás, cerrado.
    match /{todo=**} {
      allow read, write: if false;
    }
  }
}
